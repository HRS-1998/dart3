# 快速开始

## 安装

```
//使用内部源
npm i ct-dart3
```

## 引入

```js
import {
  Input,
  ListTempItem,
  ListTemp,
  Permission,
  More-condition
  Search,
  SearchItem,
  Http,
  Toolbar,
  Table,
  TableCloumn,
  tTextareaContent,
  Tinymce,
  TransferTree

} from 'ct-dart3';
app.use(Input);
app.use(Input);
app.use(ListTempItem);
app.use(ListTemp);
app.use(Permission);
app.use(More-condition);
app.use(Search);
app.use(SearchItem);
app.use(Http);
app.use(Toolbar);
app.use(Table);
app.use(TableCloumn);
app.use(tTextareaContent);
app.use(Tinymce);
app.use(TransferTree);
```
